\section{Manifest-Indexed Semantic Register (Proposed V2)}
\label{sec:semantic-register-manifest-v2-proposed}

\paragraph{Authority boundary.}
This proposed successor leaves every value and result issued under the current
appendix unchanged.  It supplies no transport theorem from an older register
and authorizes no live audit while either manifest below remains proposed.

\begin{definition}[Finite semantic-audit manifest]
A semantic-audit manifest \(\mathcal A\) fixes a finite universe-level set,
public seed grammar, rank bound, finite context/substitution/witness carriers,
Q0 normalizer, Q1 family identity, finite Q2 presentation grammar, and an
origin-cutoff Q3 registry contract.  For a verified public history \(H\),
\[
  \mathsf{RawSch}^{\mathcal A}_{2}(H)
\]
is the finite set of canonical schema-derivation trees generated by that
manifest.  Arbitrary well-typed ambient terms are not members merely by being
formable.
\end{definition}

\begin{definition}[Manifest-indexed family quotient]
Let \(\mathrm{nf}^{0}_{\mathcal A,H}\) be the typed Q0 normalizer and let
\(E_{\mathcal A,H}\) be the least equivalence closure of every exhaustively
decided Q1, Q2, and Q3 edge on the fixed Q0-normalized vertex set.  Define
\[
  \mathsf{Sch}^{\mathcal A}_{2}(H)
  :=
  \pi_0\!\left(
    \mathrm{nf}^{0}_{\mathcal A,H}
      (\mathsf{RawSch}^{\mathcal A}_{2}(H)),
    E_{\mathcal A,H}
  \right).
\]
A conforming verifier proves Q0 soundness and completeness separately from
Q1--Q3 finite graph-component completeness.
\end{definition}

\begin{definition}[Manifest-indexed marginal value]
For a conservative exact extension \(x\), typed weakening and its restriction
retraction induce
\[
  \mathrm{wk}^{\mathcal A}_x:
  \mathsf{Sch}^{\mathcal A}_{2}(H)
  \longrightarrow
  \mathsf{Sch}^{\mathcal A}_{2}(I(H,x)).
\]
The marginal family set and value are
\[
  \operatorname{Marg}^{\mathcal A}_2(H;x)
  :=
  \mathsf{Sch}^{\mathcal A}_{2}(I(H,x))
  \setminus
  \operatorname{im}(\mathrm{wk}^{\mathcal A}_x),
  \qquad
  \nu^{\mathcal A}_H(x)
  :=
  \left|\operatorname{Marg}^{\mathcal A}_2(H;x)\right|.
\]
Support-touching is a required theorem about every member of this exact set,
not an additional filter.
\end{definition}

\begin{definition}[Separate kernel-cost manifest]
A cost manifest \(\mathcal K\) fixes the public-clause grammar, public versus
ambient availability judgment, dependency graph, finite free-completion
rules, complete negative-decision procedure, and presentation quotient.
An admissible basis \(S\) reconstructs the exact public API from the
predecessor and pre-existing demand contracts, is independent clause by
clause, and is unique up to the adopted presentation equivalence.  Then
\[
  \mathsf K^{\mathcal K}_H(x):=S,
  \qquad
  \kappa^{\mathcal K}_H(x):=|S|.
\]
An absent completion proof is not an irreducibility proof.  Multiple
inequivalent bases, missing negative evidence, or an incomplete clause
disposition leaves the audit undefined.
\end{definition}

\begin{definition}[Bound value identity and comparability]
Every issued pair binds:
\[
  (\operatorname{digest}\mathcal A,
   \operatorname{digest}\mathcal K,
   \operatorname{digest}\mathrm{GF2Kernel},
   \operatorname{digest}\mathrm{Normalizer},
   \operatorname{digest}H,
   \operatorname{digest}Q3_H).
\]
Values entering one ratio have identical audit, cost, kernel, and normalizer
protocol identities.  Their act-local history identities form a certified
contiguous boundary chain, and their Q3 cutoffs obey the same empty-registry
contract or an explicit compatible restriction theorem.  Byte-identical
history digests are neither required nor generally possible across acts.
\end{definition}

\begin{definition}[Fail-closed audit]
Every source tuple, auxiliary witness, normalized vertex pair, public clause,
weakening class, and provenance assignment receives a total disposition.
Only a certified inapplicability excludes a tuple.  An outside-fragment input,
unknown disposition, failed restriction retraction, ambiguous provenance,
provenance collision, or incomplete negative certificate leaves the act
undefined; none is converted into zero.
\end{definition}
